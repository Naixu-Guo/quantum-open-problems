% Search for numbered problem files up to this inclusive cutoff.  Missing file
% numbers are skipped, so gaps in the problem pool do not stop later inputs.
\newcommand{\problemfilecutoff}{1000}
\newcount\problemfileindex
\newcommand{\inputproblemfiles}{%
  \begingroup
  \problemfileindex=1\relax
  \loop
    \IfFileExists{problem_pool/problem_\the\problemfileindex.tex}{%
      {\small\input{problem_pool/problem_\the\problemfileindex.tex}}%
    }{}%
    \advance\problemfileindex by 1\relax
  \ifnum\problemfileindex<\numexpr\problemfilecutoff+1\relax
  \repeat
  \endgroup
}
